    22→    22→async function searchAddress(query: string): Promise<NominatimResult[]> {
    23→    23→  if (!query || query.length < 3) return []
    24→    24→  try {
    25→    25→    const res = await fetch(
    26→    26→      `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&countrycodes=bo&limit=5`,
    27→    27→      { headers: { 'Accept-Language': 'es' } }
    28→    28→    )
    29→    29→    if (!res.ok) return []
    30→    30→    return await res.json()
    31→    31→  } catch {
    32→    32→    return []
    33→    33→  }
    34→    34→}

    37→    37→async function calculateRoute(
    38→    38→  origin: LatLng,
    39→    39→  destination: LatLng,
    40→    40→  stops: LatLng[]
    41→    41→): Promise<{ distance: number; duration: number; geometry: [number, number][] } | null> {
    42→    42→  const coords = [origin, ...stops, destination]
    43→    43→    .map(c => `${c.lng},${c.lat}`)
    44→    44→    .join(';')
    45→    45→  try {
    46→    46→    const res = await fetch(
    47→    47→      `https://router.project-osrm.org/route/v1/driving/${coords}?overview=full&steps=true`
    48→    48→    )
    49→    49→    if (!res.ok) return null
    50→    50→    const data = await res.json()
    51→    51→    if (!data.routes || data.routes.length === 0) return null
    52→    52→    const route = data.routes[0]
    53→    53→    const geometry: [number, number][] = []
    54→    54→    if (route.geometry) {
    55→    55→      if (typeof route.geometry === 'string') {
    56→    56→        // Decode polyline
    57→    57→        let index = 0
    58→    58→        let lat = 0
    59→    59→        let lng = 0
    60→    60→        while (index < (route.geometry as string).length) {
    61→    61→          let byte: number
    62→    62→          let shift = 0
    63→    63→          let result = 0
    64→    64→          do {
    65→    65→            byte = (route.geometry as string).charCodeAt(index++) - 63
    66→    66→            result |= (byte & 0x1f) << shift
    67→    67→            shift += 5
    68→    68→          } while (byte >= 0x20)
    69→    69→          const dlat = ((result & 1) !== 0 ? ~(result >> 1) : (result >> 1))
    70→    70→          lat += dlat
    71→    71→          shift = 0
    72→    72→          result = 0
    73→    73→          do {
    74→    74→            byte = (route.geometry as string).charCodeAt(index++) - 63
    75→    75→            result |= (byte & 0x1f) << shift
    76→    76→            shift += 5
    77→    77→          } while (byte >= 0x20)
    78→    78→          const dlng = ((result & 1) !== 0 ? ~(result >> 1) : (result >> 1))
    79→    79→          lng += dlng
    80→    80→          geometry.push([lat / 1e5, lng / 1e5])
    81→    81→        }
    82→    82→      } else if (Array.isArray(route.geometry)) {
    83→    83→        for (const coord of route.geometry) {
    84→    84→          if (Array.isArray(coord) && coord.length >= 2) {
    85→    85→            geometry.push([coord[1], coord[0]])
    86→    86→          }
    87→    87→        }
    88→    88→      }
    89→    89→    }
    90→    90→    return {
    91→    91→      distance: route.distance / 1000,
    92→    92→      duration: route.duration / 60,
    93→    93→      geometry,
    94→    94→    }
    95→    95→  } catch {
    96→    96→    return null
    97→    97→  }
    98→    98→}

   101→   101→function useInView(threshold = 0.1) {
   102→   102→  const ref = useRef<HTMLDivElement>(null)
   103→   103→  const [visible, setVisible] = useState(false)
   104→   104→  useEffect(() => {
   105→   105→    const el = ref.current
   106→   106→    if (!el) return
   107→   107→    const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) setVisible(true) }, { threshold })
   108→   108→    obs.observe(el)
   109→   109→    return () => obs.disconnect()
   110→   110→  }, [threshold])
   111→   111→  return { ref, visible }
   112→   112→}
   113→   113→
   114→   114→function A({ children, className = '', delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   115→   115→  const { ref, visible } = useInView(0.08)
   116→   116→  return (
   117→   117→    <div ref={ref} className={`transition-all duration-700 ease-out ${visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'} ${className}`}
   118→   118→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}>{children}</div>
   119→   119→  )
   120→   120→}

   703→   703→        }
   704→   704→      }

   707→   707→        rem -= largest.cap
   708→   708→      }
   709→   709→    }

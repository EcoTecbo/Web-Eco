    15→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    16→function useInView(threshold = 0.1) {
    17→  const ref = useRef<HTMLDivElement>(null)
    18→  const [visible, setVisible] = useState(false)
    19→  useEffect(() => {
    20→    const el = ref.current
    21→    if (!el) return
    22→    const obs = new IntersectionObserver(
    23→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    24→      { threshold }
    25→    )
    26→    obs.observe(el)
    27→    return () => obs.disconnect()
    28→  }, [threshold])
    29→  return { ref, visible }
    30→}

    43→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}

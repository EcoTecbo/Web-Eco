    17→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    18→function useInView(threshold = 0.1) {
    19→  const ref = useRef<HTMLDivElement>(null)
    20→  const [visible, setVisible] = useState(false)
    21→  useEffect(() => {
    22→    const el = ref.current
    23→    if (!el) return
    24→    const obs = new IntersectionObserver(
    25→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    26→      { threshold }
    27→    )
    28→    obs.observe(el)
    29→    return () => obs.disconnect()
    30→  }, [threshold])
    31→  return { ref, visible }
    32→}
    33→
    34→function AnimatedSection({ children, className = '', delay = 0 }: {
    35→  children: React.ReactNode; className?: string; delay?: number
    36→}) {
    37→  const { ref, visible } = useInView(0.08)
    38→  return (
    39→    <div
    40→      ref={ref}
    41→      className={`transition-all duration-700 ease-out ${
    42→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
    43→      } ${className}`}
    44→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}
    45→    >
    46→      {children}
    47→    </div>
    48→  )
    49→}

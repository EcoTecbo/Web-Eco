    21→/* Nominatim search */
    22→async function searchAddress(query: string): Promise<NominatimResult[]> {
    23→  if (!query || query.length < 3) return []
    24→  try {
    25→    const res = await fetch(
    26→      `https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(query)}&countrycodes=bo&limit=5`,
    27→      { headers: { 'Accept-Language': 'es' } }
    28→    )
    29→    if (!res.ok) return []
    30→    return await res.json()
    31→  } catch {
    32→    return []
    33→  }
    34→}

    36→/* OSRM route calculation */
    37→async function calculateRoute(
    38→  origin: LatLng,
    39→  destination: LatLng,
    40→  stops: LatLng[]
    41→): Promise<{ distance: number; duration: number; geometry: [number, number][] } | null> {
    42→  const coords = [origin, ...stops, destination]
    43→    .map(c => `${c.lng},${c.lat}`)
    44→    .join(';')
    45→  try {
    46→    const res = await fetch(
    47→      `https://router.project-osrm.org/route/v1/driving/${coords}?overview=full&steps=true`
    48→    )
    49→    if (!res.ok) return null
    50→    const data = await res.json()
    51→    if (!data.routes || data.routes.length === 0) return null
    52→    const route = data.routes[0]
    53→    const geometry: [number, number][] = []
    54→    if (route.geometry) {
    55→      if (typeof route.geometry === 'string') {
    56→        // Decode polyline
    57→        let index = 0
    58→        let lat = 0
    59→        let lng = 0
    60→        while (index < (route.geometry as string).length) {
    61→          let byte: number
    62→          let shift = 0
    63→          let result = 0
    64→          do {
    65→            byte = (route.geometry as string).charCodeAt(index++) - 63
    66→            result |= (byte & 0x1f) << shift
    67→            shift += 5
    68→          } while (byte >= 0x20)
    69→          const dlat = ((result & 1) !== 0 ? ~(result >> 1) : (result >> 1))
    70→          lat += dlat
    71→          shift = 0
    72→          result = 0
    73→          do {
    74→            byte = (route.geometry as string).charCodeAt(index++) - 63
    75→            result |= (byte & 0x1f) << shift
    76→            shift += 5
    77→          } while (byte >= 0x20)
    78→          const dlng = ((result & 1) !== 0 ? ~(result >> 1) : (result >> 1))
    79→          lng += dlng
    80→          geometry.push([lat / 1e5, lng / 1e5])
    81→        }
    82→      } else if (Array.isArray(route.geometry)) {
    83→        for (const coord of route.geometry) {
    84→          if (Array.isArray(coord) && coord.length >= 2) {
    85→            geometry.push([coord[1], coord[0]])
    86→          }
    87→        }
    88→      }
    89→    }
    90→    return {
    91→      distance: route.distance / 1000,
    92→      duration: route.duration / 60,
    93→      geometry,
    94→    }
    95→  } catch {
    96→    return null
    97→  }
    98→}

   101→function useInView(threshold = 0.1) {
   102→  const ref = useRef<HTMLDivElement>(null)
   103→  const [visible, setVisible] = useState(false)
   104→  useEffect(() => {
   105→    const el = ref.current
   106→    if (!el) return
   107→    const obs = new IntersectionObserver(([e]) => { if (e.isIntersecting) setVisible(true) }, { threshold })
   108→    obs.observe(el)
   109→    return () => obs.disconnect()
   110→  }, [threshold])
   111→  return { ref, visible }
   112→}

   313→function HeroSection() {
   314→  const truckRef = useRef<HTMLDivElement>(null)
   315→  const [heroSlide, setHeroSlide] = useState(0)
   316→  const heroImages = ['/mudanza-hero1.webp', '/mudanza-hero2.webp']
   317→
   318→  useEffect(() => {
   319→    if (truckRef.current) {
   320→      anime({ targets: truckRef.current, translateX: [-120, 0], opacity: [0, 1], duration: 1400, easing: 'easeOutExpo' })
   321→    }
   322→  }, [])
   323→
   324→  // Auto-slide hero images
   325→  useEffect(() => {
   326→    const interval = setInterval(() => {
   327→      setHeroSlide(prev => (prev + 1) % heroImages.length)
   328→    }, 5000)
   329→    return () => clearInterval(interval)
   330→  }, [heroImages.length])
   331→
   332→  return (
   333→    <section className="relative min-h-[92vh] flex items-center justify-center overflow-hidden">
   334→      {/* Background images with fade transition */}
   335→      {heroImages.map((img, i) => (
   336→        <div key={img} className="absolute inset-0 bg-cover bg-center transition-opacity duration-1000"

   732→        additional.push(largest)

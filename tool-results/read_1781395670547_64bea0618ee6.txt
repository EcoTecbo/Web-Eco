    19→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    20→function useInView(threshold = 0.1) {
    21→  const ref = useRef<HTMLDivElement>(null)
    22→  const [visible, setVisible] = useState(false)
    23→  useEffect(() => {
    24→    const el = ref.current
    25→    if (!el) return
    26→    const obs = new IntersectionObserver(
    27→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    28→      { threshold }
    29→    )
    30→    obs.observe(el)
    31→    return () => obs.disconnect()
    32→  }, [threshold])
    33→  return { ref, visible }
    34→}

    36→function AnimatedSection({ children, className = '', delay = 0 }: {
    37→  children: React.ReactNode; className?: string; delay?: number
    38→}) {
    39→  const { ref, visible } = useInView(0.08)
    40→  return (
    41→    <div
    42→      ref={ref}
    43→      className={`transition-all duration-700 ease-out ${
    44→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
    45→      } ${className}`}
    46→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}
    47→    >
    48→      {children}
    49→    </div>
    50→  )
    51→}

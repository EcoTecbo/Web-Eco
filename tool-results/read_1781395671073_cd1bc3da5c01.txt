    16→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    17→function useInView(threshold = 0.1) {
    18→  const ref = useRef<HTMLDivElement>(null)
    19→  const [visible, setVisible] = useState(false)
    20→  useEffect(() => {
    21→    const el = ref.current
    22→    if (!el) return
    23→    const obs = new IntersectionObserver(
    24→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    25→      { threshold }
    26→    )
    27→    obs.observe(el)
    28→    return () => obs.disconnect()
    29→  }, [threshold])
    30→  return { ref, visible }
    31→}

    34→function AnimatedSection({ children, className = '', delay = 0 }: {
    35→  children: React.ReactNode; className?: string; delay?: number
    36→}) {
    37→  const { ref, visible } = useInView(0.08)
    38→  return (
    39→    <div
    40→      ref={ref}
    41→      className={`transition-all duration-700 ease-out ${
    42→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
    43→      } ${className}`}
    44→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}
    45→    >
    46→      {children}
    47→    </div>
    48→  )
    49→}

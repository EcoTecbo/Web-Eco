    17→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    18→function useInView(threshold = 0.1) {
    19→  const ref = useRef<HTMLDivElement>(null)
    20→  const [visible, setVisible] = useState(false)
    21→  useEffect(() => {
    22→    const el = ref.current
    23→    if (!el) return
    24→    const obs = new IntersectionObserver(
    25→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    26→      { threshold }
    27→    )
    28→    obs.observe(el)
    29→    return () => obs.disconnect()
    30→  }, [threshold])
    31→  return { ref, visible }
    32→}

    34→/* ─────────────────────── animated section wrapper ─────────────────────────── */
    35→function AnimatedSection({ children, className = '', delay = 0 }: {
    36→  children: React.ReactNode; className?: string; delay?: number
    37→}) {
    38→  const { ref, visible } = useInView(0.08)
    39→  return (
    40→    <div
    41→      ref={ref}
    42→      className={`transition-all duration-700 ease-out ${
    43→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
    44→      } ${className}`}
    45→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}
    46→    >
    47→      {children}
    48→    </div>
    49→  )
    50→}

    15→/* ─────────────────────── scroll-triggered animation hook ──────────────────── */
    16→function useInView(threshold = 0.1) {
    17→  const ref = useRef<HTMLDivElement>(null)
    18→  const [visible, setVisible] = useState(false)
    19→  useEffect(() => {
    20→    const el = ref.current
    21→    if (!el) return
    22→    const obs = new IntersectionObserver(
    23→      ([e]) => { if (e.isIntersecting) setVisible(true) },
    24→      { threshold }
    25→    )
    26→    obs.observe(el)
    27→    return () => obs.disconnect()
    28→  }, [threshold])
    29→  return { ref, visible }
    30→}

    32→/* ─────────────────────── animated section wrapper ─────────────────────────── */
    33→function AnimatedSection({ children, className = '', delay = 0 }: {
    34→  children: React.ReactNode; className?: string; delay?: number
    35→}) {
    36→  const { ref, visible } = useInView(0.08)
    37→  return (
    38→    <div
    39→      ref={ref}
    40→      className={`transition-all duration-700 ease-out ${
    41→        visible ? 'opacity-100 translate-y-0' : 'opacity-0 translate-y-10'
    42→      } ${className}`}
    43→      style={{ transitionDelay: visible ? `${delay}ms` : '0ms' }}
    44→    >
    45→      {children}
    46→    </div>
    47→  )
    48→}
